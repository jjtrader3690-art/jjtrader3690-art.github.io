                showLoader();

                if (e.target.matches('[data-id][data-uid][data-amount]')) { const depositId = e.target.dataset.id; const userId = e.target.dataset.uid; const amount = parseFloat(e.target.dataset.amount); const userRef = db.collection('users').doc(userId); const depositRef = db.collection('deposits').doc(depositId); try { await db.runTransaction(async (transaction) => { transaction.update(depositRef, { status: 'Completed' }); transaction.update(userRef, { walletBalance: firebase.firestore.FieldValue.increment(amount) }); transaction.update(userRef.collection('deposits').doc(depositId), { status: 'Completed' }); }); showAlert('Deposit approved!'); await renderDeposits(); } catch (error) { showAlert('Error: ' + error.message); } } 
                else if (e.target.matches('.btn-danger[data-id]') && e.target.closest('#deposits-table-body')) { const depositId = e.target.dataset.id; try { const depositDoc = await db.collection('deposits').doc(depositId).get(); const userId = depositDoc.data().uid; const batch = db.batch(); batch.update(db.collection('deposits').doc(depositId), { status: 'Rejected' }); batch.update(db.collection('users').doc(userId).collection('deposits').doc(depositId), { status: 'Rejected' }); await batch.commit(); showAlert('Deposit rejected.'); await renderDeposits(); } catch (error) { showAlert('Error: ' + error.message); } }
                // Order Actions
                else if (e.target.matches('.btn-success[data-id]') && e.target.closest('#orders-table-body')) { const orderId = e.target.dataset.id; try { const orderDoc = await db.collection('orders').doc(orderId).get(); const userId = orderDoc.data().uid; const batch = db.batch(); batch.update(db.collection('orders').doc(orderId), { status: 'Completed' }); batch.update(db.collection('users').doc(userId).collection('orders').doc(orderId), { status: 'Completed' }); await batch.commit(); showAlert('Order marked as complete.'); await renderOrders(); } catch (error) { showAlert('Error: ' + error.message); } } 
                else if (e.target.matches('.btn-danger[data-id][data-uid]') && e.target.closest('#orders-table-body')) { const orderId = e.target.dataset.id; const userId = e.target.dataset.uid; const price = parseFloat(e.target.dataset.price); const userRef = db.collection('users').doc(userId); const orderRef = db.collection('orders').doc(orderId); try { await db.runTransaction(async (transaction) => { transaction.update(orderRef, { status: 'Rejected' }); transaction.update(userRef.collection('orders').doc(orderId), { status: 'Rejected' }); transaction.update(userRef, { walletBalance: firebase.firestore.FieldValue.increment(price) }); }); showAlert('Order rejected and amount refunded.'); await renderOrders(); } catch (error) { showAlert('Error: ' + error.message); } }
                // Product Actions
                else if (e.target.matches('#add-product-btn')) { openProductModal(); }
                else if (e.target.matches('.btn-edit-product')) { const doc = await db.collection('products').doc(e.target.dataset.id).get(); openProductModal(doc.id, doc.data()); }
                else if (e.target.matches('.btn-delete-product')) { if (confirm('Are you sure?')) { await db.collection('products').doc(e.target.dataset.id).delete(); showAlert('Product deleted.'); await renderProducts(); } }
                hideLoader();
